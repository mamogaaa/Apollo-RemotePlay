cmake_minimum_required(VERSION 3.20)
project(remoteplay_windows C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../build")
set(CMAKE_WINDOWS_EXPORT_ALL_SYMBOLS ON)
find_package(OpenSSL REQUIRED)
get_filename_component(RP "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
add_library(rpcrypt SHARED "${RP}/gkbridge.c")
foreach(src rpcrypt gkcrypt log thread time)
  target_sources(rpcrypt PRIVATE "${CHIAKI_SOURCE}/lib/src/${src}.c")
endforeach()
target_include_directories(rpcrypt PRIVATE "${CHIAKI_SOURCE}/lib/include")
target_link_libraries(rpcrypt PRIVATE OpenSSL::Crypto)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory("${RP}/../../third-party/moonlight-common-c" moonlight)
add_library(moonbridge SHARED "${RP}/moonbridge.c")
target_link_libraries(moonbridge PRIVATE moonlight-common-c)
set_target_properties(rpcrypt moonbridge PROPERTIES PREFIX "")
